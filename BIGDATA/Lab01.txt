--- A1-A3: pobranie i rozpakowanie danych ---
mkdir -p ~/data ~/downloads
cd ~/data
wget https://files.grouplens.org/datasets/movielens/ml-100k.zip
unzip ml-100k.zip
ls ~/data/ml-100k-MM
cat ~/data/ml-100k-MM/README

--- 0.1 katalogi w HDFS ---
hdfs dfs -ls /user/maria_dev

--- 0.2 utworzenie katalogu ---
hdfs dfs -mkdir -p /user/maria_dev/ASSESSMENT/ml100k

--- 0.3 kopiowanie u.data i u.item do HDFS ---
hdfs dfs -put ~/data/ml-100k-MM/u.data ~/data/ml-100k-MM/u.item /user/maria_dev/ASSESSMENT/ml100k

--- 0.4 weryfikacja ---
hdfs dfs -ls /user/maria_dev/ASSESSMENT/ml100k/

--- 0.5 pierwsze 10 wierszy u.data ---
hdfs dfs -cat /user/maria_dev/ASSESSMENT/ml100k/u.data | head -n 10

--- 0.6 kopia w obrebie HDFS ---
hdfs dfs -mkdir -p /user/maria_dev/movielens/copy
hdfs dfs -cp /user/maria_dev/ASSESSMENT/ml100k/u.data /user/maria_dev/movielens/copy/

--- 0.7 pobranie u.item z HDFS do ~/downloads ---
hdfs dfs -get /user/maria_dev/ASSESSMENT/ml100k/u.item ~/downloads/

--- 0.9 rozmiar katalogu (wykonac PRZED 0.8) ---
hdfs dfs -du -h /user/maria_dev/movielens/copy/
hdfs dfs -du -s -h /user/maria_dev/movielens/copy/

--- 0.8 usuniecie katalogu ---
hdfs dfs -rm -r /user/maria_dev/movielens/copy

--- 0.10 przygotowanie danych (katalog copy odtworzony po usunieciu) ---
hdfs dfs -mkdir -p /user/maria_dev/movielens/copy
hdfs dfs -cp /user/maria_dev/ASSESSMENT/ml100k/u.data /user/maria_dev/movielens/copy/
hdfs dfs -cat /user/maria_dev/movielens/copy/* > all_data.txt
hdfs dfs -put all_data.txt /user/maria_dev/movielens/copy/
hdfs dfs -ls /user/maria_dev/movielens/copy/
